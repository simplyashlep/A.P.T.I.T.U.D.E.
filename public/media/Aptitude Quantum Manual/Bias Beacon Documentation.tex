	\section{Architecture Independence Declaration}
	
	This specification governs the Beacon Analytics Service — the institutional scoring pipeline of Aptitude Systems. Before any dimensional specification is read, the following invariant must be understood and enforced throughout implementation:
	
	\begin{invariantbox}
		\textbf{\sffamily INVARIANT B-0: INSTITUTIONAL INDEPENDENCE}
		
		\medskip
		The Beacon scoring system is architecturally independent of individual user activity. Aptitude Scores are derived from public institutional records, regulatory filings, court decisions, and agency-generated data. Individual incident submissions from the Doc~1 pipeline are a \emph{supplementary signal source} — one channel among many — and their absence does not degrade institutional score validity. An institution cannot suppress, dilute, or game its Aptitude Score by discouraging individual filings. \textbf{The public record is the primary and sufficient evidentiary basis.}
	\end{invariantbox}
	
	\bigskip
	This independence is enforced at the data layer: the \texttt{behavioral\_quanta} table accepts a \texttt{source\_type} field. Only records with:
	
	\begin{lstlisting}
		source_type IN (
		'public_record', 'regulatory_filing', 'court_decision',
		'agency_report', 'cfpb_complaint', 'ftc_action',
		'inspector_general'
		)
	\end{lstlisting}
	
	\noindent are eligible for institutional score computation. Records with \texttt{source\_type = 'individual\_submission'} are tagged as supplementary and weighted at $\leq 5\%$ of any dimensional score unless corroborated by a public record.

	\section{Composite Scoring Formula}
	
	\subsection{Aptitude Score \texorpdfstring{$(A)$}{(A)}}
	
	The Aptitude Score is a weighted linear combination of the six dimensional scores:
	
	\[
	A = \sum_{i=1}^{6} w_i \cdot D_i \quad \text{where } D_i \in \{F, C, T, E, Q, Ac\}
	\]
	\[
	\text{subject to: } \sum_{i=1}^{6} w_i = 1.0, \quad w_i > 0 \; \forall i
	\]
	
	\medskip
	\noindent Default AHP-derived weights, configurable per institution class:
	
	\bigskip
	\begin{center}
		\begin{tabular}{L{3.8cm} C{1.4cm} C{2.4cm} L{6.0cm}}
			\toprule
			\rowcolor{navy}\color{white}\textbf{\sffamily Dimension} & \color{white}\textbf{\sffamily Symbol} & \color{white}\textbf{\sffamily Default Weight} & \color{white}\textbf{\sffamily Rationale} \\
			\midrule
			Fidelity       & $F$   & $w_1 = 0.22$ & Legal mandate adherence is foundational \\
			\rowcolor{aptlight}
			Consistency    & $C$   & $w_2 = 0.18$ & Variance in outcomes signals systemic bias \\
			Transparency   & $T$   & $w_3 = 0.17$ & Public record access is a legal right \\
			\rowcolor{aptlight}
			Efficiency     & $E$   & $w_4 = 0.15$ & Resource stewardship proxies competence \\
			Equity         & $Q$   & $w_5 = 0.16$ & Demographic fairness is a constitutional floor \\
			\rowcolor{aptlight}
			Accountability & $Ac$  & $w_6 = 0.12$ & Self-correction capacity determines trajectory \\
			\bottomrule
		\end{tabular}
	\end{center}
	
	\subsection{Min-Max Normalization}
	
	All sub-metric raw values are normalized before dimensional score computation:
	
	\[
	D_{\text{norm}} = \frac{D_{\text{raw}} - D_{\min}}{D_{\max} - D_{\min}}
	\]
	
	Bounds are configurable per institution class (see Section~\ref{sec:cra-bounds}). Values are clamped to $[0, 1]$ — no extrapolation permitted. Normalized values are stored in \texttt{behavioral\_quanta.magnitude}.

	\section{Dimension 1: Fidelity \texorpdfstring{$(\Phi_F)$}{}}
	
	\begin{dimensionbox}
		\textbf{\sffamily FIDELITY} — Measures whether an institution adheres to its legal mandate, enabling statutes, and constitutional obligations. The fidelity field collapses to zero when an institution acts outside statutory authority, systematically ignores regulatory obligations, or applies law selectively.
	\end{dimensionbox}
	
	\subsection{Dimensional Formula}
	
	\[
	F = w_1 \cdot \text{LMCR} + w_2 \cdot \text{PAS} + w_3 \cdot \text{RVR} + w_4 \cdot (1 - \text{SOF})
	\]
	
	\noindent where LMCR = Legal Mandate Compliance Rate; PAS = Policy Adherence Score; RVR = Regulatory Violation Reversal Rate; SOF = Statutory Override Frequency (inverted).
	
	\subsection{Sub-Metric Specification}
	
	\begin{longtable}{L{1.1cm} L{1.9cm} L{4.5cm} L{3.5cm} C{1.1cm} C{1.1cm}}
		\toprule
		\rowcolor{navy}\color{white}\textbf{\sffamily Code} & \color{white}\textbf{\sffamily Name} & \color{white}\textbf{\sffamily Definition} & \color{white}\textbf{\sffamily Source} & \color{white}\textbf{\sffamily Range} & \color{white}\textbf{\sffamily Wt.} \\
		\midrule
		\endhead
		\texttt{LMCR} & Legal Mandate Compliance Rate & Ratio of actions complying with enabling statute to total actions. Computed from regulatory findings, court orders, and IG reports. & CFPB exams; OCC findings; State AG filings; Court orders & $[0,1]$ & 0.30 \\
		\midrule
		\rowcolor{aptlight}\texttt{PAS} & Policy Adherence Score & Ratio of internal policy decisions aligning with promulgated regulations. Derived from enforcement action records. & FTC actions; CFPB consent orders; Regulatory correspondence & $[0,1]$ & 0.25 \\
		\midrule
		\texttt{RVR} & Regulatory Violation Reversal Rate & Fraction of identified violations corrected before external enforcement. Self-correction elevates fidelity; forced correction does not. & IG reports; Voluntary corrective action filings; Audit records & $[0,1]$ & 0.25 \\
		\midrule
		\rowcolor{aptlight}\texttt{SOF} & Statutory Override Frequency & Rate at which institution actions required courts or regulators to enforce statutory limits. Inverted in formula. & Federal/state court dockets; Regulatory enforcement database & $[0,1]$ & 0.20 \\
		\bottomrule
	\end{longtable}
	
	\subsection{Agent Specifications}
	
	\begin{agentbox}
		\textbf{\sffamily\color{navy} AGENT: Fidelity Signal Parser} — Mandate Compliance Extraction
		
		\medskip
		\textbf{Inputs:} Raw regulatory filings, court orders, IG reports, consent orders (JSON/PDF)
		
		\textbf{Outputs:} Structured fidelity quanta with LMCR/PAS/RVR/SOF magnitude per action
		
		\textbf{Invariants:} ZERO tolerance for uncited compliance claims. Every quantum must carry \texttt{source\_id}, \texttt{document\_date}, and \texttt{jurisdiction\_code}. Agent \textsc{must} flag ambiguous compliance characterizations for human review — it does not resolve them. Rejects any document older than 7 years unless it constitutes ongoing pattern evidence.
	\end{agentbox}
	
	\medskip
	
	\begin{agentbox}
		\textbf{\sffamily\color{navy} AGENT: Fidelity Field Synthesizer} — Dimensional Score Computation
		
		\medskip
		\textbf{Inputs:} Normalized fidelity quanta for target institution and time window
		
		\textbf{Outputs:} $F$ score $\in [0,1]$ with confidence interval, quantum count, source breakdown, and outlier flags
		
		\textbf{Invariants:} Minimum 12 quanta required for score computation. If quantum count $< 12$, emit \texttt{INSUFFICIENT\_DATA} status — do not estimate. Outliers ($|x - \mu| > 2\sigma$) are retained but flagged. Score must be reproducible byte-for-byte from the quantum set.
	\end{agentbox}
	
	\subsection{Dimensional Invariants}
	
	\begin{itemize}[leftmargin=1.5em]
		\item $F$ score requires a minimum of 12 behavioral quanta from at least 3 distinct source types.
		\item Consent orders count as negative compliance evidence regardless of whether the institution "agreed" to them.
		\item Self-reported compliance documentation receives a 0.50 credibility discount unless corroborated by independent regulatory finding.
		\item A single court finding of \emph{ultra vires} action triggers an automatic LMCR floor of 0.20 for the affected period regardless of other quanta.
	\end{itemize}

	\section{Dimension 2: Consistency \texorpdfstring{$(\Phi_C)$}{}}
	
	\begin{dimensionbox}
		\textbf{\sffamily CONSISTENCY} — Measures variance in institutional outcomes across substantively similar cases. Inconsistency is the statistical signature of bias, favoritism, and arbitrary decision-making. Consistency measures whether like cases are treated alike — not whether outcomes are correct.
	\end{dimensionbox}
	
	\subsection{Dimensional Formula}
	
	\[
	C = w_1 \cdot (1 - \text{OVI}) + w_2 \cdot \text{PAR} + w_3 \cdot (1 - \text{CDS}) + w_4 \cdot (1 - \text{DRR})
	\]
	
	\subsection{Sub-Metric Specification}
	
	\begin{longtable}{L{1.1cm} L{1.9cm} L{4.5cm} L{3.5cm} C{1.1cm} C{1.1cm}}
		\toprule
		\rowcolor{navy}\color{white}\textbf{\sffamily Code} & \color{white}\textbf{\sffamily Name} & \color{white}\textbf{\sffamily Definition} & \color{white}\textbf{\sffamily Source} & \color{white}\textbf{\sffamily Range} & \color{white}\textbf{\sffamily Wt.} \\
		\midrule
		\endhead
		\texttt{OVI} & Outcome Variance Index & Statistical variance in outcomes across substantively similar cases scored by the Legal Classifier Agent. \textit{Inverted in formula.} & Case outcome databases; FOIA decision records; Public dockets & $[0,1]$ & 0.30 \\
		\midrule
		\rowcolor{aptlight}\texttt{PAR} & Precedent Adherence Rate & Rate at which current decisions align with the institution's own prior decisions in analogous cases. & Administrative law records; Internal guidance; Published decisions & $[0,1]$ & 0.25 \\
		\midrule
		\texttt{CDS} & Cross-Case Disparity Score & Magnitude of outcome differences between demographically distinct but legally identical case populations. \textit{Inverted.} & HMDA data; CFPB demographic reports; Fair lending exams; EEOC filings & $[0,1]$ & 0.25 \\
		\midrule
		\rowcolor{aptlight}\texttt{DRR} & Decision Reversal Rate & Rate at which institution decisions are overturned on appeal or administrative review. \textit{Inverted.} & Federal/state appellate records; Administrative review outcomes & $[0,1]$ & 0.20 \\
		\bottomrule
	\end{longtable}
	
	\subsection{Agent Specifications}
	
	\begin{agentbox}
		\textbf{\sffamily\color{navy} AGENT: Case Similarity Engine} — Substantive Equivalence Classification
		
		\medskip
		\textbf{Inputs:} Raw case records with outcome data, demographic indicators, and procedural history
		
		\textbf{Outputs:} Case clusters with similarity scores; OVI and CDS raw values per cluster
		
		\textbf{Invariants:} Similarity scoring uses semantic embedding (\texttt{VECTOR(1536)}) + structured attribute matching. Demographic attributes are isolated variables — they must \textsc{not} appear as similarity factors. Engine must expose its similarity rationale for every cluster. Minimum cluster size = 30 cases before OVI is computed.
	\end{agentbox}
	
	\medskip
	
	\begin{agentbox}
		\textbf{\sffamily\color{navy} AGENT: Consistency Auditor} — Variance and Reversal Computation
		
		\medskip
		\textbf{Inputs:} Case clusters, precedent database, appellate outcome records
		
		\textbf{Outputs:} $C$ score $\in [0,1]$ with per-sub-metric breakdown, cluster report, and disparity flags
		
		\textbf{Invariants:} Any sub-metric below 0.30 triggers \texttt{CONSISTENCY\_ALERT} appended to the score record. Reversal data must be sourced from appellate court directly — institution self-reporting of reversal rates is not accepted. CDS computation must be independently reproduced for any Tier IV or V classification.
	\end{agentbox}
	
	\subsection{Dimensional Invariants}
	
	\begin{itemize}[leftmargin=1.5em]
		\item OVI is computed at the case-cluster level. A single high-variance cluster is published separately without suppressing the overall score.
		\item Demographic disparity above 15\% in any protected class triggers an \textsc{equity cross-flag} regardless of overall $C$ score.
		\item Institutions under court-ordered desegregation or consent decrees carry those records as permanent low-consistency markers until a 5-year clean record is established.
		\item Selective retroactive application of policy changes is penalized; uniform prospective changes are not.
	\end{itemize}

	\section{Dimension 3: Transparency \texorpdfstring{$(\Phi_T)$}{}}
	
	\begin{dimensionbox}
		\textbf{\sffamily TRANSPARENCY} — Measures whether the institution makes its records, decisions, and procedures accessible. Transparency is not a courtesy — for most public institutions it is a legal requirement. The transparency field penalizes stonewalling, delay, redaction abuse, and procedural opacity with the same weight as outright violation.
	\end{dimensionbox}
	
	\subsection{Dimensional Formula}
	
	\[
	T = w_1 \cdot \text{RDR} + w_2 \cdot \text{RTI} + w_3 \cdot \text{FCR} + w_4 \cdot \text{PAX}
	\]
	
	\subsection{Sub-Metric Specification}
	
	\begin{longtable}{L{1.1cm} L{1.9cm} L{4.5cm} L{3.5cm} C{1.1cm} C{1.1cm}}
		\toprule
		\rowcolor{navy}\color{white}\textbf{\sffamily Code} & \color{white}\textbf{\sffamily Name} & \color{white}\textbf{\sffamily Definition} & \color{white}\textbf{\sffamily Source} & \color{white}\textbf{\sffamily Range} & \color{white}\textbf{\sffamily Wt.} \\
		\midrule
		\endhead
		\texttt{RDR} & Record Disclosure Rate & Fraction of requested records fully disclosed without substantively contested redactions. Partial disclosures count proportionally. & FOIA.gov; State public records databases; Disclosure litigation records & $[0,1]$ & 0.28 \\
		\midrule
		\rowcolor{aptlight}\texttt{RTI} & Response Time Index & $\min(\text{deadline}/\text{median\_actual}, 1.0)$. Faster than statutory deadline $= 1.0$; slower degrades proportionally. & FOIA logs cross-validated with requester data; Court dockets & $[0,1]$ & 0.25 \\
		\midrule
		\texttt{FCR} & FOIA/Public Records Compliance Rate & Rate of requests fulfilled without requiring administrative appeal or litigation. Legally compelled disclosures are counted as non-compliant. & DOJ OIP FOIA statistics; State AG reports; Litigation outcomes & $[0,1]$ & 0.27 \\
		\midrule
		\rowcolor{aptlight}\texttt{PAX} & Proactive Access Index & Rate at which institution voluntarily publishes decision rationales, policy changes, and statistical reports without being requested. Benchmarked against peer class. & Agency public portals; Federal Register; State rule publication logs & $[0,1]$ & 0.20 \\
		\bottomrule
	\end{longtable}
	
	\subsection{Agent Specifications}
	
	\begin{agentbox}
		\textbf{\sffamily\color{navy} AGENT: Transparency Crawler} — Public Record Availability Mapping
		
		\medskip
		\textbf{Inputs:} Agency identifiers, known public portals, FOIA log databases, court docket APIs
		
		\textbf{Outputs:} Transparency quanta per agency with RDR/RTI/FCR/PAX raw values and citation chain
		
		\textbf{Invariants:} Agent must verify record availability from a non-institutional network path. Availability that requires login, payment, or special permission is treated as restricted. Re-crawl every 30 days. Any FOIA litigation loss automatically sets FCR quantum $= 0.0$ for the contested record category for 24 months.
	\end{agentbox}
	
	\medskip
	
	\begin{agentbox}
		\textbf{\sffamily\color{navy} AGENT: Transparency Adjudicator} — Redaction and Delay Scoring
		
		\medskip
		\textbf{Inputs:} Disclosure records, response timelines, redaction justification codes
		
		\textbf{Outputs:} $T$ score $\in [0,1]$; redaction abuse flag; delay pattern analysis; proactive publication inventory
		
		\textbf{Invariants:} Deliberative process privilege and FOIA Exemption 5 are the most abused exemptions — any invocation triggers heightened scrutiny. Agent must crosscheck claimed exemption against successful appeal rates for that exemption at the same agency. Pattern of exemption abuse ($>25\%$ appeal overturn rate) generates a \texttt{REDACTION\_ABUSE} flag that reduces RDR by 0.15.
	\end{agentbox}
	
	\subsection{Dimensional Invariants}
	
	\begin{itemize}[leftmargin=1.5em]
		\item Response time is measured from the date of institutional \emph{receipt}, not date of acknowledgment — the clock cannot be reset by delaying acknowledgment.
		\item Institutions requiring in-person or physical mail submission only receive an automatic PAX deduction of 0.10.
		\item Sealed settlement agreements containing public interest matters are treated as non-disclosures for RDR purposes.
		\item An institution held in contempt for disclosure failure within the last 5 years cannot score above $T = 0.40$ regardless of subsequent performance.
	\end{itemize}

	\section{Dimension 4: Efficiency \texorpdfstring{$(\Phi_E)$}{}}
	
	\begin{dimensionbox}
		\textbf{\sffamily EFFICIENCY} — Measures how effectively an institution converts public resources — funding, staffing, authority — into mandated outcomes. Efficiency is not austerity: an institution that is underfunded but maximizes what it has scores high. Efficiency is the closest proxy to institutional competence.
	\end{dimensionbox}
	
	\subsection{Dimensional Formula}
	
	\[
	E = w_1 \cdot \text{CRR} + w_2 \cdot \text{PTS} + w_3 \cdot \text{RUI} + w_4 \cdot \text{BCR}
	\]
	
	\subsection{Sub-Metric Specification}
	
	\begin{longtable}{L{1.1cm} L{1.9cm} L{4.5cm} L{3.5cm} C{1.1cm} C{1.1cm}}
		\toprule
		\rowcolor{navy}\color{white}\textbf{\sffamily Code} & \color{white}\textbf{\sffamily Name} & \color{white}\textbf{\sffamily Definition} & \color{white}\textbf{\sffamily Source} & \color{white}\textbf{\sffamily Range} & \color{white}\textbf{\sffamily Wt.} \\
		\midrule
		\endhead
		\texttt{CRR} & Case Resolution Rate & Fraction of cases resolved within the applicable statutory or regulatory time limit. Administrative closures without documented outcome do \emph{not} count. & Agency performance reports; Congressional testimony; GAO/IG audits & $[0,1]$ & 0.28 \\
		\midrule
		\rowcolor{aptlight}\texttt{PTS} & Processing Time Score & $\min(\text{benchmark\_median}/\text{agency\_median}, 1.0)$. Normalized against peer-class institutional benchmark and statutory limit. & Annual reports; FOIA case management logs; Regulatory performance standards & $[0,1]$ & 0.25 \\
		\midrule
		\texttt{RUI} & Resource Utilization Index & Ratio of measurable mandate-advancing outcomes to annual operating budget. \textit{Net of litigation costs.} Normalized against peer class. & Agency budget justifications; Congressional appropriations; Published performance metrics & $[0,1]$ & 0.22 \\
		\midrule
		\rowcolor{aptlight}\texttt{BCR} & Backlog Clearance Rate & Year-over-year backlog trajectory. Growing $< 0.50$; stable $= 0.50$; reducing $> 0.50$. & Annual reports; IG backlog findings; Congressional oversight testimony & $[0,1]$ & 0.25 \\
		\bottomrule
	\end{longtable}
	
	\subsection{Agent Specifications}
	
	\begin{agentbox}
		\textbf{\sffamily\color{navy} AGENT: Efficiency Metrics Extractor} — Performance Data Ingestion
		
		\medskip
		\textbf{Inputs:} Agency performance reports, budget justifications, GAO/IG audits, congressional testimony
		
		\textbf{Outputs:} Efficiency quanta with CRR/PTS/RUI/BCR values, fiscal year tagged, peer-class benchmarked
		
		\textbf{Invariants:} Peer-class benchmarking is \textsc{mandatory}. The agent maintains a peer-class registry keyed by \texttt{institution\_type} $\times$ \texttt{budget\_tier} $\times$ \texttt{jurisdiction\_level}. If peer class has fewer than 5 members, emit \texttt{PEER\_CLASS\_INSUFFICIENT} and do not score.
	\end{agentbox}
	
	\medskip
	
	\begin{agentbox}
		\textbf{\sffamily\color{navy} AGENT: Efficiency Trend Analyzer} — Longitudinal Performance Assessment
		
		\medskip
		\textbf{Inputs:} Efficiency quanta across minimum 3 fiscal years, budget change records
		
		\textbf{Outputs:} $E$ score $\in [0,1]$; trend vector (improving/stable/degrading); resource shock flags
		
		\textbf{Invariants:} Budget cuts exceeding 20\% in a single year trigger a \texttt{BUDGET\_SHOCK} flag applying a documented 0.10 grace adjustment to $E$ for that year only. Institutions showing degrading $E$ scores despite stable or increasing budgets receive a \texttt{MANAGEMENT\_EFFICIENCY} flag surfaced in the public score breakdown.
	\end{agentbox}
	
	\subsection{Dimensional Invariants}
	
	\begin{itemize}[leftmargin=1.5em]
		\item Administrative closures without documented outcome are flagged as \texttt{ADMINISTRATIVE\_CLOSURE} and do not count toward CRR.
		\item Efficiency scores are always displayed with the peer-class benchmark visible to users.
		\item RUI is computed net of litigation costs — funds spent defending against the institution's own violations are excluded from the outcome-advancing numerator.
		\item Courts-ordered capacity constraints are documented as annotations — the constraint is noted but does not excuse backlogs.
	\end{itemize}

	\section{Dimension 5: Equity \texorpdfstring{$(\Phi_Q)$}{}}
	
	\begin{dimensionbox}
		\textbf{\sffamily EQUITY} — Measures whether the institution delivers its mandate fairly across demographic groups, geographic areas, and economic strata. Equity is the constitutional floor of institutional legitimacy. An institution that scores perfectly on every other dimension but systematically disadvantages protected classes fails the public trust at the most fundamental level.
	\end{dimensionbox}
	
	\subsection{Dimensional Formula}
	
	\[
	Q = w_1 \cdot (1 - \text{DDI}) + w_2 \cdot \text{GAS} + w_3 \cdot \text{PCOR} + w_4 \cdot \text{ASR}
	\]
	
	\subsection{Sub-Metric Specification}
	
	\begin{longtable}{L{1.1cm} L{1.9cm} L{4.5cm} L{3.5cm} C{1.1cm} C{1.1cm}}
		\toprule
		\rowcolor{navy}\color{white}\textbf{\sffamily Code} & \color{white}\textbf{\sffamily Name} & \color{white}\textbf{\sffamily Definition} & \color{white}\textbf{\sffamily Source} & \color{white}\textbf{\sffamily Range} & \color{white}\textbf{\sffamily Wt.} \\
		\midrule
		\endhead
		\texttt{DDI} & Demographic Disparity Index & Composite index of outcome rate differences across race, gender, age, disability, and national origin for substantively similar cases. \textit{Inverted in formula.} & HMDA; EEOC charge data; CFPB fair lending exams; Disparate impact litigation & $[0,1]$ & 0.30 \\
		\midrule
		\rowcolor{aptlight}\texttt{GAS} & Geographic Access Score & Service availability and outcome quality across rural, suburban, and urban geographies. Includes physical, digital, and language access. & USPS ZIP-level data; Agency field office maps; Language access compliance reports & $[0,1]$ & 0.23 \\
		\midrule
		\texttt{PCOR} & Protected Class Outcome Rate & Ratio of favorable outcomes for protected class members to the expected rate at parity, controlling for legally relevant variables. Score $= 1.0$ at parity. & Civil rights investigation outcomes; Fair lending exam results; Title VI/VII enforcement & $[0,1]$ & 0.27 \\
		\midrule
		\rowcolor{aptlight}\texttt{ASR} & Appeals Success Rate (Protected Class) & Rate at which protected-class complainants who challenge adverse decisions prevail on appeal. High rates indicate systematic erroneous adverse decisions. & Federal/state appellate records; Administrative review outcomes; Civil rights findings & $[0,1]$ & 0.20 \\
		\bottomrule
	\end{longtable}
	
	\subsection{Agent Specifications}
	
	\begin{agentbox}
		\textbf{\sffamily\color{navy} AGENT: Equity Signal Extractor} — Demographic Disparity Detection
		
		\medskip
		\textbf{Inputs:} Case outcome records with demographic indicators, geographic data, HMDA/EEOC/CFPB filings
		
		\textbf{Outputs:} DDI and PCOR values per demographic dimension; disparity alerts; statistical significance flags
		
		\textbf{Invariants:} Minimum $n = 100$ per demographic cell before statistical claims are made. Cells below $n = 100$ are excluded and documented as \texttt{INSUFFICIENT\_SAMPLE}. Agent \textsc{never} imputes demographic data. Any disparity reaching significance at $p < 0.05$ triggers \texttt{EQUITY\_ALERT} regardless of DDI magnitude.
	\end{agentbox}
	
	\medskip
	
	\begin{agentbox}
		\textbf{\sffamily\color{navy} AGENT: Equity Field Enforcer} — Score Computation with Hard Floors
		
		\medskip
		\textbf{Inputs:} Equity quanta, disparity alerts, litigation records, consent decree status
		
		\textbf{Outputs:} $Q$ score $\in [0,1]$; mandatory flag set; hard-floor enforcement log; public disclosure summary
		
		\textbf{Hard Floors (override weighted average — cannot be bypassed):}
		\begin{itemize}[leftmargin=1.5em]
			\item Finding of intentional discrimination within 10 years $\Rightarrow Q_{\min} = 0.15$
			\item Active consent decree for civil rights violation $\Rightarrow Q_{\min} = 0.25$
			\item Pattern-or-practice finding $\Rightarrow Q_{\min} = 0.20$
		\end{itemize}
		All floors are documented in the score record. Hard floors cannot be overridden by composite score computation.
	\end{agentbox}
	
	\subsection{Dimensional Invariants}
	
	\begin{itemize}[leftmargin=1.5em]
		\item Equity is the only dimension with mandatory hard floors that override the weighted average.
		\item An institution may submit a legally-relevant variable model to justify disparity — that model must be validated by the Signal Harmonizer Agent before it is accepted.
		\item Geographic access scoring includes digital-only alternatives only if 90\%+ of the service population has documented reliable broadband access.
		\item Language access failures are treated as geographic access failures — a physically present but linguistically inaccessible office scores no higher than an absent one.
		\item A Tier V equity score triggers a mandatory regulatory referral workflow regardless of composite Aptitude Score.
	\end{itemize}

	\section{Dimension 6: Accountability \texorpdfstring{$(\Phi_{Ac})$}{}}
	
	\begin{dimensionbox}
		\textbf{\sffamily ACCOUNTABILITY} — Measures the institution's capacity and willingness to identify its own failures, correct them, and institutionalize reforms that prevent recurrence. Accountability is the temporal dimension — it captures trajectory, not just state. It is the dimension most predictive of future behavior.
	\end{dimensionbox}
	
	\subsection{Dimensional Formula}
	
	\[
	Ac = w_1 \cdot \text{SCR} + w_2 \cdot \text{AC} + w_3 \cdot \text{CRR}_{\text{comp}} + w_4 \cdot \text{RAR}
	\]
	
	\noindent where SCR = Self-Correction Rate; AC = Audit Compliance; $\text{CRR}_{\text{comp}}$ = Complaint Resolution Rate (quality measure, distinct from Efficiency CRR); RAR = Remediation Action Rate.
	
	\subsection{Sub-Metric Specification}
	
	\begin{longtable}{L{1.1cm} L{1.9cm} L{4.5cm} L{3.5cm} C{1.1cm} C{1.1cm}}
		\toprule
		\rowcolor{navy}\color{white}\textbf{\sffamily Code} & \color{white}\textbf{\sffamily Name} & \color{white}\textbf{\sffamily Definition} & \color{white}\textbf{\sffamily Source} & \color{white}\textbf{\sffamily Range} & \color{white}\textbf{\sffamily Wt.} \\
		\midrule
		\endhead
		\texttt{SCR} & Self-Correction Rate & Rate at which institution identifies and corrects errors through its own review processes before external enforcement, litigation, or complaint compels action. Proactive correction receives $1.5\times$ multiplier. & Voluntary corrective action filings; Annual report disclosures; IG management responses & $[0,1]$ & 0.28 \\
		\midrule
		\rowcolor{aptlight}\texttt{AC} & Audit Compliance & Rate of compliance with internal audit recommendations, regulatory examination findings, and IG recommendations within the mandated response window. Tracks \emph{implementation}, not acknowledgment. & IG semiannual reports; GAO follow-up audits; Regulatory examination tracking & $[0,1]$ & 0.25 \\
		\midrule
		$\texttt{CRR}_{\texttt{c}}$ & Complaint Resolution Rate & Rate of consumer/public complaints resolved to documented satisfaction within the institution's stated timeline. Audited by 10\% random sampling of "resolved" complaints. & CFPB complaint database; BBB records; State AG databases; Agency complaint logs (FOIA) & $[0,1]$ & 0.22 \\
		\midrule
		\rowcolor{aptlight}\texttt{RAR} & Remediation Action Rate & Fraction of committed remediation actions implemented on the committed timeline. Delayed implementation scores as partial credit proportional to delay magnitude. & Consent order monitoring reports; Court-appointed monitor reports; IG follow-up findings & $[0,1]$ & 0.25 \\
		\bottomrule
	\end{longtable}
	
	\subsection{Agent Specifications}
	
	\begin{agentbox}
		\textbf{\sffamily\color{navy} AGENT: Self-Audit Intelligence Agent} — Correction Origin Classification
		
		\medskip
		\textbf{Inputs:} Corrective action records, enforcement timelines, complaint databases, audit response logs
		
		\textbf{Outputs:} SCR and RAR values with correction-origin classification (self-initiated vs.\ externally compelled) and timeline analysis
		
		\textbf{Invariants:} Classification is binary: self-initiated (SCR credit) or externally compelled (no SCR credit). A correction triggered by a complaint that the institution received but had not yet acted on for $>30$ days is classified as externally compelled. Agent must maintain a correction provenance chain for every SCR quantum — the origin must be independently verifiable.
	\end{agentbox}
	
	\medskip
	
	\begin{agentbox}
		\textbf{\sffamily\color{navy} AGENT: Accountability Enforcement Agent} — Commitment vs.\ Performance Tracking
		
		\medskip
		\textbf{Inputs:} Consent order commitments, audit recommendation logs, remediation schedules, IG monitoring reports
		
		\textbf{Outputs:} $Ac$ score $\in [0,1]$; commitment fulfillment ledger; chronic non-compliance flags; trajectory vector
		
		\textbf{Invariants:} \texttt{CHRONIC\_ACCOUNTABILITY\_FAILURE} flag is set if $AC < 0.40$ for 3 consecutive scoring periods. Flag persists for 5 years and is published in the public score breakdown. Remediation promises without documented implementation score as zero. Resolution fraud detection: 10\% random sample of "resolved" complaints is traced; fabricated resolutions generate \texttt{RESOLUTION\_FRAUD} flag.
	\end{agentbox}
	
	\subsection{Dimensional Invariants}
	
	\begin{itemize}[leftmargin=1.5em]
		\item SCR receives a $1.5\times$ multiplier for corrections made proactively before any complaint or audit raised the issue.
		\item A pattern of recurring identical violations (same type in 3+ consecutive periods) automatically caps $Ac$ at 0.35.
		\item Institutions in receivership, under mandatory monitor, or under a corporate integrity agreement are scored against their compliance schedule, not voluntary benchmarks.
		\item Institutions that mark complaints resolved without documented resolution receive a \texttt{RESOLUTION\_FRAUD} flag that triggers an audit of all CRR scores for the preceding 24 months.
	\end{itemize}
